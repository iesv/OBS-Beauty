cmake_minimum_required(VERSION 3.28)
if(APPLE)
  set(CMAKE_OSX_ARCHITECTURES arm64 CACHE STRING "Apple Silicon only" FORCE)
  set(CMAKE_OSX_DEPLOYMENT_TARGET 12.0 CACHE STRING "Minimum macOS version")
endif()
project(OBS-Beauty VERSION 1.0.0 LANGUAGES CXX)
if(NOT (WIN32 OR APPLE) OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Supported targets: Windows x64 and macOS arm64.")
endif()
set(OBS_ROOT "" CACHE PATH "OBS Studio 32.0.4 runtime directory or OBS.app")
set(OBS_SOURCE_DIR "" CACHE PATH "OBS Studio 32.0.4 source tree")
option(BUILD_TESTING "Build the libobs rendering and stress tests" ON)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
if(NOT EXISTS "${OBS_SOURCE_DIR}/libobs/obs-module.h")
  message(FATAL_ERROR "OBS_SOURCE_DIR must contain the OBS Studio 32.0.4 headers.")
endif()
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/obs-sdk")
set(OBS_RELEASE_CANDIDATE 0)
set(OBS_BETA 0)
configure_file("${OBS_SOURCE_DIR}/libobs/obsconfig.h.in" "${CMAKE_CURRENT_BINARY_DIR}/obs-sdk/obsconfig.h")
add_library(OBS::libobs SHARED IMPORTED)
set_target_properties(OBS::libobs PROPERTIES
  INTERFACE_INCLUDE_DIRECTORIES "${OBS_SOURCE_DIR}/libobs;${CMAKE_CURRENT_BINARY_DIR}/obs-sdk")
if(WIN32)
  if(NOT MSVC)
    message(FATAL_ERROR "Use MSVC x64 on Windows.")
  endif()
  if(NOT EXISTS "${OBS_ROOT}/bin/64bit/obs.dll")
    message(FATAL_ERROR "OBS_ROOT must contain bin/64bit/obs.dll.")
  endif()
  get_filename_component(msvc_bin "${CMAKE_CXX_COMPILER}" DIRECTORY)
  find_program(OBS_DUMPBIN dumpbin HINTS "${msvc_bin}" REQUIRED)
  find_program(OBS_LIB_TOOL lib HINTS "${msvc_bin}" REQUIRED)
  execute_process(COMMAND "${OBS_DUMPBIN}" /nologo /exports "${OBS_ROOT}/bin/64bit/obs.dll"
    OUTPUT_VARIABLE obs_exports COMMAND_ERROR_IS_FATAL ANY)
  string(REPLACE "\n" ";" obs_export_lines "${obs_exports}")
  set(obs_def "LIBRARY obs.dll\nEXPORTS\n")
  set(obs_export_count 0)
  foreach(line IN LISTS obs_export_lines)
    if(line MATCHES "^[ \t]+[0-9]+[ \t]+[0-9A-Fa-f]+[ \t]+[0-9A-Fa-f]+[ \t]+([A-Za-z_][A-Za-z_0-9]*)")
      string(APPEND obs_def "${CMAKE_MATCH_1}\n")
      math(EXPR obs_export_count "${obs_export_count}+1")
    endif()
  endforeach()
  if(obs_export_count LESS 100)
    message(FATAL_ERROR "Cannot read the OBS API exports.")
  endif()
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/obs-sdk/obs.def" "${obs_def}")
  execute_process(COMMAND "${OBS_LIB_TOOL}" /nologo /machine:x64
    "/def:${CMAKE_CURRENT_BINARY_DIR}/obs-sdk/obs.def" "/out:${CMAKE_CURRENT_BINARY_DIR}/obs-sdk/obs.lib"
    OUTPUT_QUIET COMMAND_ERROR_IS_FATAL ANY)
  set_target_properties(OBS::libobs PROPERTIES
    IMPORTED_IMPLIB "${CMAKE_CURRENT_BINARY_DIR}/obs-sdk/obs.lib"
    IMPORTED_LOCATION "${OBS_ROOT}/bin/64bit/obs.dll")
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreadedDLL")
  set(package_platform windows-x64)
else()
  set(obs_frameworks "${OBS_ROOT}/Contents/Frameworks")
  find_library(OBS_LIBRARY NAMES libobs obs PATHS "${obs_frameworks}" NO_DEFAULT_PATH REQUIRED)
  if(IS_DIRECTORY "${OBS_LIBRARY}")
    set(OBS_LIBRARY "${OBS_LIBRARY}/libobs")
  endif()
  find_path(SIMDE_INCLUDE_DIR simde/x86/sse2.h PATHS /opt/homebrew/include REQUIRED)
  set_target_properties(OBS::libobs PROPERTIES IMPORTED_LOCATION "${OBS_LIBRARY}")
  set_property(TARGET OBS::libobs APPEND PROPERTY INTERFACE_INCLUDE_DIRECTORIES "${SIMDE_INCLUDE_DIR}")
  set(package_platform macos-arm64)
endif()
add_custom_target(clean-dist
  COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/CleanDist.cmake" VERBATIM)
add_library(obs-beauty MODULE src/beauty-filter.cpp)
add_dependencies(obs-beauty clean-dist)
target_link_libraries(obs-beauty PRIVATE OBS::libobs)
target_compile_features(obs-beauty PRIVATE cxx_std_17)
target_compile_definitions(obs-beauty PRIVATE PLUGIN_VERSION="${PROJECT_VERSION}")
set_target_properties(obs-beauty PROPERTIES PREFIX "")
if(WIN32)
  target_link_libraries(obs-beauty PRIVATE user32)
  target_compile_options(obs-beauty PRIVATE /W4 /WX /utf-8 /permissive-)
  target_link_options(obs-beauty PRIVATE /Brepro /INCREMENTAL:NO /DEBUG:NONE)
  install(TARGETS obs-beauty RUNTIME DESTINATION obs-beauty/bin/64bit LIBRARY DESTINATION obs-beauty/bin/64bit)
  install(DIRECTORY data/ DESTINATION obs-beauty/data)
  install(FILES LICENSE DESTINATION obs-beauty)
  install(FILES scripts/install_OBS-Beauty.bat DESTINATION .)
else()
  target_link_libraries(obs-beauty PRIVATE "-framework CoreGraphics")
  target_compile_options(obs-beauty PRIVATE -Wall -Wextra -Werror)
  set_target_properties(obs-beauty PROPERTIES SUFFIX ""
    BUILD_WITH_INSTALL_RPATH TRUE INSTALL_RPATH "@executable_path/../Frameworks")
  configure_file(cmake/Info.plist.in "${CMAKE_CURRENT_BINARY_DIR}/Info.plist" @ONLY)
  install(TARGETS obs-beauty LIBRARY DESTINATION obs-beauty.plugin/Contents/MacOS)
  install(FILES "${CMAKE_CURRENT_BINARY_DIR}/Info.plist" DESTINATION obs-beauty.plugin/Contents)
  install(DIRECTORY data/ DESTINATION obs-beauty.plugin/Contents/Resources)
  install(FILES LICENSE DESTINATION obs-beauty.plugin/Contents/Resources)
  install(PROGRAMS scripts/install_OBS-Beauty.sh DESTINATION .)
  # libobs recognizes the .plugin path and loads the extensionless bundle binary.
  set(test_plugin "${CMAKE_CURRENT_BINARY_DIR}/package/obs-beauty.plugin/Contents/MacOS/obs-beauty")
endif()
add_custom_target(package-dist ALL
  COMMAND "${CMAKE_COMMAND}" "-DBUILD_DIRECTORY=${CMAKE_CURRENT_BINARY_DIR}" "-DCONFIG=$<CONFIG>"
    "-DPLUGIN_VERSION=${PROJECT_VERSION}" "-DPACKAGE_PLATFORM=${package_platform}"
    "-DPYTHON_EXECUTABLE=${Python3_EXECUTABLE}" -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/Package.cmake"
  DEPENDS obs-beauty VERBATIM)
if(BUILD_TESTING)
  enable_testing()
  add_executable(beauty-integration tests/integration.cpp)
  add_dependencies(beauty-integration clean-dist)
  target_link_libraries(beauty-integration PRIVATE OBS::libobs)
  target_compile_features(beauty-integration PRIVATE cxx_std_17)
  if(WIN32)
    target_link_libraries(beauty-integration PRIVATE Psapi)
    target_compile_options(beauty-integration PRIVATE /W4 /WX /utf-8 /permissive-)
  else()
    enable_language(OBJCXX)
    target_sources(beauty-integration PRIVATE tests/mac-main.mm)
    target_link_libraries(beauty-integration PRIVATE "-framework AppKit" "-framework OpenGL")
    target_compile_options(beauty-integration PRIVATE -Wall -Wextra -Werror)
    set_target_properties(beauty-integration PROPERTIES BUILD_RPATH "${obs_frameworks}")
  endif()
  if(DEFINED ENV{GITHUB_ACTIONS})
    set(functional_args --functional)
  endif()
  if(APPLE AND DEFINED ENV{GITHUB_ACTIONS})
    target_compile_definitions(beauty-integration PRIVATE BEAUTY_CI_STRESS)
  endif()
  if(WIN32)
    set(test_plugin "$<TARGET_FILE:obs-beauty>")
  endif()
  add_test(NAME beauty-integration COMMAND beauty-integration
    "${test_plugin}" "${CMAKE_CURRENT_SOURCE_DIR}/data" "${OBS_ROOT}" ${functional_args})
  add_test(NAME beauty-stress COMMAND beauty-integration
    "${test_plugin}" "${CMAKE_CURRENT_SOURCE_DIR}/data" "${OBS_ROOT}" --stress)
  if(WIN32)
    set_tests_properties(beauty-integration beauty-stress PROPERTIES
      ENVIRONMENT_MODIFICATION "PATH=path_list_prepend:${OBS_ROOT}/bin/64bit")
  else()
    set_tests_properties(beauty-integration beauty-stress PROPERTIES
      ENVIRONMENT_MODIFICATION "DYLD_FRAMEWORK_PATH=path_list_prepend:${obs_frameworks}")
  endif()
  set_tests_properties(beauty-integration PROPERTIES TIMEOUT 180)
  if(APPLE)
    # Keep pixel regressions separate from the longer software-rendering stress test.
    set_tests_properties(beauty-integration PROPERTIES TIMEOUT 60)
  endif()
  set_tests_properties(beauty-stress PROPERTIES TIMEOUT 600 LABELS stress)
endif()
